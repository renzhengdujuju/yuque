# 定义常量：

static+final定义常量

-常量：在程序运行期间，值不可以发生改变的量，我们称之为常量

```java
package com.eagles.test;

public class UserTest {

    private String name;
    private String address;
    private int age;

    public UserTest(String name, String address, int age) {
        this.name = name;
        this.address = address;
        this.age = age;
    }

    public UserTest() {
    }
    //再创建一个无参构造是为了防止其他文件在调用原来的无参构造，因为有参构造会覆盖无参构造
    //国籍
    // static的特点：类加载的时候才会加载，只加载一次 （大写＋_）
    private final static String COUNTRY_NAME = "中国";

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getAddress() {
        return address;
    }

    public void setAddress(String address) {
        this.address = address;
    }

    public int getAge() {
        return age;
    }

    public void setAge(int age) {
        this.age = age;
    }

    public boolean equals(Object obj) {

        if (this == obj) {
            return true;
        }

        //如果用户对象的姓名、年龄和居住地址完全一样的情况下，就返回true
        if (obj != null && obj instanceof UserTest) {
            //向下转型
            UserTest user = (UserTest) obj;
            if (user.getName().equals(this.getName()) && user.getAge() == this.getAge() && user.getAddress().equals(this.getAddress())) {
                return true;
            }
        }

        return false;
    }

    public void main(String[] args) {

        //        User user1 = new User();
        //        User user2 = new User();
        //        User user3 = new User();
        //        User user4 = new User();
        //        User user5 = new User();
        //        User user6 = new User();
        //        User user7 = new User();

    }

}
```
